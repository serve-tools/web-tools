    import { connect } from "@serve-tools/client-shared-db/scope/window";
import type { DemoSchema } from "./database-worker.js";

const worker = new SharedWorker(new URL("./database-worker.ts", import.meta.url), {
	type: "module",
});
const database = connect<DemoSchema>(worker.port);

const refresh = async (revision: number) => {
	const tasks = await database.getAll("tasks");
	taskList.textContent = tasks.map((task) => task.title).join(", ");
	revisionOutput.textContent = String(revision);
};

const subscription = database.subscribe("tasks", (change) => {
	void refresh(change.revision);
}, {
	onReady: (revision) => void refresh(revision),
});

await database.add("tasks", {
	done: false,
	id: crypto.randomUUID(),
	title: titleInput.value,
	updatedAt: Date.now(),
});

subscription.unsubscribe();
database.close();
