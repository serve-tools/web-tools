@serve-tools/client-input

Input has a beginning and an end.

Inspect normalized pointer geometry and drag sessions without surrendering browser defaults.

Press and drag

Waiting for a primary pointer.

Core code; element references and local types match the example above.

import { observePointer } from "@serve-tools/client-input";

const stop = observePointer(pointerPad, {
	start(state, event) {
		if (!event.isPrimary || event.button !== 0) return false;
		pointerPad.toggleAttribute("data-active", true);
	},
	move(state) {
		pointerPad.style.setProperty("--pointer-x", `${state.ratio.x * 100}%`);
		pointerPad.style.setProperty("--pointer-y", `${state.ratio.y * 100}%`);
	},
	end() {
		pointerPad.removeAttribute("data-active");
	},
});

addEventListener("pagehide", stop, { once: true });

import { observeDropTarget } from "@serve-tools/client-input";

const stop = observeDropTarget(dropZone, {
	start: () => dropZone.toggleAttribute("data-active", true),
	over(event) {
		if (event.dataTransfer?.types.includes("text/plain")) {
			event.preventDefault();
		}
	},
	end: () => dropZone.removeAttribute("data-active"),
});

addEventListener("pagehide", stop, { once: true });
