@serve-tools/client-context

Context follows the component tree.

Move one subscribing consumer between providers, then change either value. The nearest active provider owns delivery without global state or prototype patches.

Provider A

Ocean

Provider B

Sunset

The consumer stays subscribed while moving and immediately resolves against its new ancestry.

Core code; element references and local types match the example above.

import { ContextConsumer, ContextProvider, createContext } from "@serve-tools/client-context";

interface Theme { accent: string; name: string }

const theme = createContext<Theme>(Symbol("theme"));
const provider = new ContextProvider(providerElement, {
	context: theme,
	initialValue: { name: "Ocean", accent: "#087e8b" },
});
const nextProvider = new ContextProvider(nextProviderElement, {
	context: theme,
	initialValue: { name: "Sunset", accent: "#d95d39" },
});
const consumer = new ContextConsumer(consumerElement, {
	context: theme,
	subscribe: true,
	callback: (value) => {
		consumerElement.textContent = `${value.name} · ${value.accent}`;
	},
});

provider.connect();
nextProvider.connect();
consumer.connect();
provider.setValue({ name: "Tide", accent: "#2667ff" });

nextProviderElement.append(consumerElement);
consumer.refresh();

addEventListener("pagehide", () => {
	consumer.disconnect();
	provider.disconnect();
	nextProvider.disconnect();
}, { once: true });
