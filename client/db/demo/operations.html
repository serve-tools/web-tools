Opening the database…

Core code; element references and local types match the example above.

import { DB } from "@serve-tools/client-db";
import type { DemoSchema } from "./database.js";

const db = await DB.open<DemoSchema>("serve-tools-client-db-demo", {
	version: 1,
	upgrade(database) {
		const contacts = database.createObjectStore("contacts", { keyPath: "id" });

		contacts.createIndex("byEmail", "email", { unique: true });
		database.createObjectStore("activity", { autoIncrement: true });
	},
});

await db.add("contacts", {
	id: crypto.randomUUID(),
	name: nameInput.value,
	email: emailInput.value,
	visits: 0,
});
const contacts = await db.getAll("contacts");
