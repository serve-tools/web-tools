    Core code; element references and local types match the example above.

    import {
	getKeyChord,
	getKeyChordAriaKeyShortcuts,
	getKeyChordLabel,
	getKeyChordSymbols,
	matchKeyChord,
} from "@serve-tools/client-keyboard";

capture.addEventListener("keydown", (event) => {
	const chord = getKeyChord(event);
	if (!chord) return;

	if (matchKeyChord("Mod+K", event)) event.preventDefault();

	canonical.textContent = chord;
	label.textContent = getKeyChordLabel(chord);
	aria.textContent = getKeyChordAriaKeyShortcuts(chord);
	symbols.replaceChildren(...getKeyChordSymbols(chord).map((value) =>
		Object.assign(document.createElement("kbd"), { textContent: value }),
	));
});
